# MCP Tools Implementation Plan (search_file + get_file)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the raw proxy in `/mcp` with a hand-rolled JSON-RPC 2.0 MCP server exposing `search_file` and `get_file` tools backed by `GET /v1/files` and `GET /v1/files/{id}`.

**Architecture:** Stateless Streamable HTTP, POST-only (`GET` → 405). Route keeps existing rate-limit + Bearer→`SCSE_ID` auth, then dispatches JSON-RPC via a pure protocol module; tool executors (zod-validated structured args → composed filter DSL → backend fetch) live in their own module. No session store, no SSE, no new deps.

**Tech Stack:** Next.js 15 App Router route handlers, zod v4 (`z.toJSONSchema`), vitest (mock valkey + stubbed fetch), Playwright request-fixture e2e.

**Spec:** `docs/superpowers/specs/2026-08-28-mcp-tools-design.md`

**Deviation from spec (intentional):** spec's e2e "full chain with stubbed backend" is not achievable — the e2e webServer is a real Next process whose `fetch` cannot be stubbed, and a valid token requires Valkey + backend. Full `tools/call` chain is covered in unit route tests instead; e2e adds protocol-surface checks that need no backend (401 with bad token, 405 on GET).

## Global Constraints

- No new npm dependencies. No new env vars. No Valkey changes.
- Supported `protocolVersion`: `2025-06-18`, `2025-03-26`, `2024-11-05`; fallback/latest `2025-06-18`.
- JSON-RPC error codes: parse `-32700`, invalid request `-32600`, method not found `-32601`, invalid params `-32602`, internal `-32603`.
- Fixed tool-error messages (exact): `Backend unavailable.`, `Session expired — please re-authenticate.`, `File not found or you do not have access.`, `Backend error (<status>)`.
- `SCSE_ID` must never appear in any tool output.
- All values entering the filter DSL pass whitelist/regex first — no escaping logic, hostile chars impossible.
- Backend timeout: `AbortSignal.timeout(15_000)`.

```ts
// src/lib/mcp-tools.ts
import { z } from "zod";

import { env } from "@/lib/env";

export type ToolOutcome =
  | { status: "ok"; text: string }
  | { status: "isError"; text: string }
  | { status: "unknownTool" }
  | { status: "invalidParams"; message: string };

const EMAIL_RE = /^[^@\s]+@[^@\s]+\.[^@\s]+$/;
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
const EXT_RE = /^[a-z0-9]{1,10}$/;
const BACKEND_TIMEOUT_MS = 15_000;

const KINDS = ["document", "image", "video", "other"] as const;
const SOURCES = [
  "shared folder",
  "apple notes",
  "onedrive",
  "onedrive-business",
  "onedrive-personal",
  "sharepoint",
  "teams",
  "unknown",
] as const;

const searchSchema = z.object({
  search: z.string().min(1).optional(),
  extension: z.array(z.string().min(1)).optional(),
  kind: z.enum(KINDS).optional(),
  source: z.enum(SOURCES).optional(),
  owner_email: z.string().regex(EMAIL_RE).optional(),
  updated_after: z.string().regex(DATE_RE).optional(),
  updated_before: z.string().regex(DATE_RE).optional(),
  sort: z
    .enum([
      "relevance",
      "updated_at",
      "name",
      "updated_at asc",
      "updated_at desc",
      "name asc",
      "name desc",
    ])
    .optional(),
  mode: z.enum(["broad", "exact"]).optional(),
  limit: z.number().int().min(1).max(50).optional(),
  offset: z.number().int().min(0).optional(),
});

const getSchema = z.object({ id: z.string().min(1) });

function invalidParams(error: z.ZodError): ToolOutcome {
  const issue = error.issues[0];
  const path = issue?.path.join(".");
  return {
    status: "invalidParams",
    message: path ? `${path}: ${issue.message}` : (issue?.message ?? "Invalid parameters"),
  };
}

async function backendFetch(url: URL, scseId: string): Promise<Response | null> {
  try {
    return await fetch(url, {
      method: "GET",
      headers: { Cookie: `SCSE_ID=${scseId}`, "User-Agent": "frontend-mcp" },
      signal: AbortSignal.timeout(BACKEND_TIMEOUT_MS),
    });
  } catch {
    return null;
  }
}

async function execSearchFile(args: unknown, scseId: string): Promise<ToolOutcome> {
  const parsed = searchSchema.safeParse(args);
  if (!parsed.success) return invalidParams(parsed.error);
  const a = parsed.data;

  const ext = (a.extension ?? []).map((e) => e.toLowerCase().trim());
  if (a.extension?.length && ext.some((e) => !EXT_RE.test(e)))
    return { status: "invalidParams", message: "extension: invalid extension" };

  // Injection safety by construction: every value above passed a whitelist/regex.
  const parts: string[] = [];
  if (ext.length) parts.push(`extension in (${ext.map((e) => `'${e}'`).join(",")})`);
  if (a.kind) parts.push(`kind eq '${a.kind}'`);
  if (a.source) parts.push(`source eq '${a.source}'`);
  if (a.owner_email) parts.push(`owner_email eq '${a.owner_email}'`);
  if (a.updated_after) parts.push(`updated_at gte '${a.updated_after}'`);
  if (a.updated_before) parts.push(`updated_at lte '${a.updated_before}'`);

  const url = new URL("/v1/files", env.BACKEND_BASE_URL);
  if (a.search) url.searchParams.set("search", a.search);
  if (parts.length) url.searchParams.set("filter", parts.join(" and "));
  url.searchParams.set("sort", a.sort ?? "relevance");
  url.searchParams.set("mode", a.mode ?? "broad");
  url.searchParams.set("limit", String(a.limit ?? 10));
  url.searchParams.set("offset", String(a.offset ?? 0));

  const res = await backendFetch(url, scseId);
  if (!res) return { status: "isError", text: "Backend unavailable." };
  if (res.status === 200) return { status: "ok", text: await res.text() };
  if (res.status === 401)
    return { status: "isError", text: "Session expired — please re-authenticate." };
  if (res.status === 400) return { status: "isError", text: await res.text() };
  return { status: "isError", text: `Backend error (${res.status})` };
}

async function execGetFile(args: unknown, scseId: string): Promise<ToolOutcome> {
  const parsed = getSchema.safeParse(args);
  if (!parsed.success) return invalidParams(parsed.error);

  const url = new URL(
    `/v1/files/${encodeURIComponent(parsed.data.id)}`,
    env.BACKEND_BASE_URL,
  );
  const res = await backendFetch(url, scseId);
  if (!res) return { status: "isError", text: "Backend unavailable." };
  if (res.status === 200) return { status: "ok", text: await res.text() };
  if (res.status === 401)
    return { status: "isError", text: "Session expired — please re-authenticate." };
  if (res.status === 404)
    return { status: "isError", text: "File not found or you do not have access." };
  return { status: "isError", text: `Backend error (${res.status})` };
}

type ToolDef = {
  name: string;
  description: string;
  inputSchema: object;
  execute: (args: unknown, scseId: string) => Promise<ToolOutcome>;
};

const searchFileTool: ToolDef = {
  name: "search_file",
  description:
    "Search the user's documents across OneDrive, SharePoint, Teams, shared folders, and Apple Notes. Returns full JSON: hits under `value[].\\_source` (use `_source.id` with get_file) and pagination under `meta.pagination`.",
  inputSchema: z.toJSONSchema(searchSchema),
  execute: execSearchFile,
};

const getFileTool: ToolDef = {
  name: "get_file",
  description:
    "Get full metadata for one document by ID. Use an `id` from search_file results. Returns a not-found error if the file does not exist or you lack access.",
  inputSchema: z.toJSONSchema(getSchema),
  execute: execGetFile,
};

export const MCP_TOOLS: ToolDef[] = [searchFileTool, getFileTool];

export function toolsList(): Array<{ name: string; description: string; inputSchema: object }> {
  return MCP_TOOLS.map(({ name, description, inputSchema }) => ({
    name,
    description,
    inputSchema,
  }));
}

export async function callTool(
  name: string,
  args: unknown,
  scseId: string,
): Promise<ToolOutcome> {
  const tool = MCP_TOOLS.find((t) => t.name === name);
  if (!tool) return { status: "unknownTool" };
  return tool.execute(args, scseId);
}

// src/lib/mcp-protocol.ts
import { callTool, toolsList } from "@/lib/mcp-tools";

const SUPPORTED_PROTOCOL_VERSIONS = ["2025-06-18", "2025-03-26", "2024-11-05"];
const LATEST_PROTOCOL_VERSION = "2025-06-18";
const SERVER_NAME = "searchpoint-mcp";
const SERVER_VERSION = "0.1.0";

type Json = Record<string, unknown>;

export type McpOutcome =
  | { httpStatus: 200; body: Json }
  | { httpStatus: 202; body: null };

function rpcError(id: unknown, code: number, message: string): Json {
  return { jsonrpc: "2.0", id, error: { code, message } };
}

function rpcResult(id: unknown, result: unknown): Json {
  return { jsonrpc: "2.0", id, result };
}

export async function handleMcpRequest(
  raw: string,
  scseId: string,
): Promise<McpOutcome> {
  let parsed: unknown;
  try {
    parsed = JSON.parse(raw);
  } catch {
    return { httpStatus: 200, body: rpcError(null, -32700, "Parse error") };
  }

  if (
    Array.isArray(parsed) ||
    parsed === null ||
    typeof parsed !== "object" ||
    (parsed as Json).jsonrpc !== "2.0" ||
    typeof (parsed as Json).method !== "string"
  ) {
    const id =
      !Array.isArray(parsed) &&
      parsed !== null &&
      typeof parsed === "object" &&
      "id" in (parsed as Json)
        ? (parsed as Json).id
        : null;
    return { httpStatus: 200, body: rpcError(id, -32600, "Invalid Request") };
  }

  const msg = parsed as Json;
  const method = msg.method as string;
  // Notification (no id) — including notifications/initialized: nothing to answer.
  if (!("id" in msg)) return { httpStatus: 202, body: null };
  const id = msg.id;

  switch (method) {
    case "initialize": {
      const requested = (msg.params as Json | undefined)?.protocolVersion;
      const version =
        typeof requested === "string" &&
        SUPPORTED_PROTOCOL_VERSIONS.includes(requested)
          ? requested
          : LATEST_PROTOCOL_VERSION;
      return {
        httpStatus: 200,
        body: rpcResult(id, {
          protocolVersion: version,
          capabilities: { tools: { listChanged: false } },
          serverInfo: { name: SERVER_NAME, version: SERVER_VERSION },
        }),
      };
    }
    case "ping":
      return { httpStatus: 200, body: rpcResult(id, {}) };
    case "tools/list":
      return { httpStatus: 200, body: rpcResult(id, { tools: toolsList() }) };
    case "tools/call": {
      const params = (msg.params ?? {}) as Json;
      const name = typeof params.name === "string" ? params.name : "";
      try {
        const outcome = await callTool(name, params.arguments, scseId);
        if (outcome.status === "unknownTool")
          return {
            httpStatus: 200,
            body: rpcError(id, -32602, `Unknown tool: ${name}`),
          };
        if (outcome.status === "invalidParams")
          return {
            httpStatus: 200,
            body: rpcError(id, -32602, outcome.message),
          };
        return {
          httpStatus: 200,
          body: rpcResult(id, {
            content: [{ type: "text", text: outcome.text }],
            isError: outcome.status === "isError",
          }),
        };
      } catch {
        return { httpStatus: 200, body: rpcError(id, -32603, "Internal error") };
      }
    }
    default:
      return { httpStatus: 200, body: rpcError(id, -32601, "Method not found") };
  }
}
```

```ts
import { NextResponse } from "next/server";

import { env } from "@/lib/env";
import { handleMcpRequest } from "@/lib/mcp-protocol";
import { deriveIssuer } from "@/lib/oauth";
import { clientIpFromRequest, limiterForScope } from "@/lib/rate-limit";
import { getJSON, oauthKeys } from "@/lib/oauth-valkey";

export async function POST(req: Request) {
  const limiter = limiterForScope("oauth");
  const rl = await limiter.consume(clientIpFromRequest(req));
  if (!rl.success)
    return NextResponse.json(
      { error: "temporarily_unavailable" },
      {
        status: 429,
        headers: { "Retry-After": String(rl.retryAfterSeconds) },
      },
    );

  const at = req.headers.get("authorization")?.replace(/^Bearer\s+/i, "");
  const issuer = (env.OAUTH_ISSUER ?? deriveIssuer(req)).replace(/\/$/, "");
  const challenge = (reason: string) =>
    new NextResponse(null, {
      status: 401,
      headers: {
        "WWW-Authenticate": `Bearer error="${reason}", resource_metadata="${issuer}/.well-known/oauth-protected-resource"`,
      },
    });

  if (!at) return challenge("invalid_request");
  const data = await getJSON<{
    scseId: string;
    scseExp: number;
    user: unknown;
    client_id: string;
    scope: string;
    resource?: string;
  }>(oauthKeys.at(at));
  if (
    !data ||
    !data.scseId ||
    typeof data.scseExp !== "number" ||
    data.scseExp < Math.floor(Date.now() / 1000)
  )
    return challenge("invalid_token");

  const raw = await req.text();
  const outcome = await handleMcpRequest(raw, data.scseId);
  if (outcome.body === null) return new NextResponse(null, { status: 202 });
  return NextResponse.json(outcome.body, { status: outcome.httpStatus });
}

// MCP Streamable HTTP: server without SSE support MUST answer GET with 405.
// Clients health-check with POST initialize, not GET.
export function GET() {
  return new NextResponse(null, { status: 405, headers: { Allow: "POST" } });
}
```
